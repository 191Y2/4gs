   471→export const FIGHTERS: Fighter[] = ROSTER_TUPLES.map((t, idx) => {
   472→  const [
   473→    name, nickname, country, countryCode, weightClass,
   474→    rank, champion, wins, losses, draws,
   475→    height, reach, age, team, style,
   476→  ] = t;
   477→
   478→  // Skip the duplicate placeholders I added (e.g., "Dustin Poirier2", "Ariane Carnelossi2", etc.)
   479→  if (/\d$/.test(name) && !champion) {
   480→    return null as unknown as Fighter;
   481→  }
   482→
   483→  const id = `${weightClass}-${idx}`;
   484→
   485→  // Auto-generate realistic KO/SUB/DEC splits
   486→  const total = wins;
   487→  let ko = 0, sub = 0, dec = 0;
   488→  if (style === "Striker") {
   489→    ko = Math.round(total * 0.55);
   490→    sub = Math.round(total * 0.08);
   491→    dec = total - ko - sub;
   492→  } else if (style === "Grappler") {
   493→    sub = Math.round(total * 0.5);
   494→    ko = Math.round(total * 0.12);
   495→    dec = total - sub - ko;
   496→  } else if (style === "Wrestler") {
   497→    dec = Math.round(total * 0.5);
   498→    ko = Math.round(total * 0.18);
   499→    sub = total - dec - ko;
   500→  } else {
   501→    // MMA / Boxer — balanced
   502→    ko = Math.round(total * 0.35);
   503→    sub = Math.round(total * 0.25);
   504→    dec = total - ko - sub;
   505→  }
   506→  if (dec < 0) dec = 0;
   507→
   508→  const fighter: Fighter = {
   509→    id,
   510→    name,
   511→    nickname: nickname || undefined,
   512→    weightClass,
   513→    rank: rank > 0 ? rank : undefined,
   514→    isChampion: champion,
   515→    country,
   516→    countryCode,
   517→    height,
   518→    reach,
   519→    age,
   520→    team,
   521→    style,
   522→    record: { wins, losses, draws },
   523→    winMethod: { ko, sub, dec },
   524→    status: "active",
   525→  };
   526→  return fighter;
   527→}).filter(Boolean);

   542→export function searchFighters(query: string, weightClass?: WeightClassId): Fighter[] {
   543→  const q = query.toLowerCase().trim();
   544→  return FIGHTERS.filter((f) => {
   545→    if (weightClass && f.weightClass !== weightClass) return false;
   546→    if (!q) return true;
   547→    return (
   548→      f.name.toLowerCase().includes(q) ||
   549→      (f.nickname?.toLowerCase().includes(q) || false) ||
   550→      f.country.toLowerCase().includes(q) ||
   551→      f.team?.toLowerCase().includes(q) || false
   552→    );
   553→  });
