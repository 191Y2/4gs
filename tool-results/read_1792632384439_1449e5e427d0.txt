    39→export function onUniverseChange(cb: () => void): () => void {

    51→// ---------- calendar ----------
    52→// weeks run Saturday → Friday; every event happens on its week's Saturday.
    53→// The clock is PURE universe time — the real-world calendar is irrelevant.
    54→// The first fight week is Oct 17, 2026; the moment a card is completed the
    55→// next Saturday unlocks, and that rolls on FOREVER: Fight Night 1 → UFC 334
    56→// → Fight Night 2 → UFC 335… one card every Saturday, endlessly. An
    57→// unfinished card pins the clock on its own Saturday until it's run or
    58→// scrapped.
    59→export function currentSaturday(): Date {
    60→  if (state.events.length === 0) return new Date(FIRST_SATURDAY);
    61→  const latest = state.events.reduce((m, e) => (e.dateISO > m.dateISO ? e : m));
    62→  if (!latest.completed) return dateOfIso(latest.dateISO);
    63→  // card done — the next fight week unlocks instantly, forever
    64→  const next = dateOfIso(latest.dateISO);
    65→  next.setDate(next.getDate() + 7);
    66→  return next;
    67→}

    71→}

   133→function boutWeightClass(a: Fighter, b: Fighter): DivisionId {
   134→  if (a.division === b.division) return a.division;
   135→  // cross-division BELT DEFENSE: a fighter holding a belt in the other's
   136→  // division fights there — a double/triple champ booked against a
   137→  // division's contender defends that belt at that weight
   138→  const beltsA = champDivisions(a);
   139→  const beltsB = champDivisions(b);
   140→  const aDefends = beltsA.includes(b.division);
   141→  const bDefends = beltsB.includes(a.division);
   142→  if (aDefends && !bDefends) return b.division;
   143→  if (bDefends && !aDefends) return a.division;
   144→  // superfight / plain cross-division booking: heavier weight class
   145→  return (DIVISION_MAP[a.division]?.limit ?? 0) >= (DIVISION_MAP[b.division]?.limit ?? 0)
   146→    ? a.division
   147→    : b.division;
   148→}
   149→
   150→// same-division bout with a reigning champion = classic title fight. A
   151→// champ vs champ cross-division bout = SUPERFIGHT — all belts on the line
   152→// and the winner becomes a double/triple/…/8-division champ. A fighter
   153→// holding a belt in the OTHER fighter's division = cross-division belt
   154→// defense — also a title fight. Everyone in this universe fights in their
   155→// prime; a legend booked against the champ is a title fight and the belt
   156→// CAN move, same as anyone else.
   157→export function isTitleBout(a: Fighter, b: Fighter): boolean {
   158→  if (isSuperfight(a, b)) return true;
   159→  const beltsA = champDivisions(a);
   160→  const beltsB = champDivisions(b);
   161→  if (beltsA.includes(b.division) || beltsB.includes(a.division)) return true;
   162→  if (a.division === b.division) return !!(a.isChampion || b.isChampion);
   163→  return false;
   164→}

   185→// apply one universe bout to both fighters (live + replay)
   186→function applyBout(a: Fighter, b: Fighter, rec: UniverseBout) {
   187→  const resA: FightLogEntry["res"] =
   188→    rec.winner === "A" ? "W" : rec.winner === "B" ? "L" : rec.winner === "DRAW" ? "D" : "NC";
   189→  const resB: FightLogEntry["res"] =
   190→    rec.winner === "B" ? "W" : rec.winner === "A" ? "L" : rec.winner === "DRAW" ? "D" : "NC";
   191→
   192→  const base = {
   193→    ev: `${rec.evName} · ${rec.section === "main" ? "MAIN EVENT" : rec.section === "prelim" ? "PRELIMS" : "EARLY PRELIMS"}`,
   194→    date: rec.date,
   195→    m: rec.method,
   196→    det: rec.detail,
   197→    r: rec.round,
   198→    t: rec.time,
   199→    wc: rec.wc,
   200→    title: rec.title ? ("T" as const) : undefined,
   201→    sim: true as const,
   202→  };
   203→  a.log.push({ ...base, op: b.name, res: resA });
   204→  b.log.push({ ...base, op: a.name, res: resB });
   205→
   206→  // OCTAGONHQ universe records (starts 0-0 — real W-L never changes)
   207→  if (resA === "W") a.simW = (a.simW ?? 0) + 1;
   208→  else if (resA === "L") a.simL = (a.simL ?? 0) + 1;
   209→  else if (resA === "D") a.simD = (a.simD ?? 0) + 1;
   210→  if (resB === "W") b.simW = (b.simW ?? 0) + 1;
   211→  else if (resB === "L") b.simL = (b.simL ?? 0) + 1;
   212→  else if (resB === "D") b.simD = (b.simD ?? 0) + 1;
   213→}

   299→export function nextEventNumber(): number {

   301→  return Math.max(FIRST_EVENT_NUMBER, max + 1);

   304→// Fight Nights carry their own infinite counter — Fight Night 1, 2, 3…
   305→export function nextFightNightNumber(): number {
   306→  let max = 0;
   307→  let count = 0;
   308→  for (const e of state.events) {
   309→    if (e.type !== "fightnight") continue;
   310→    count++;
   311→    max = Math.max(max, e.number ?? 0);
   312→  }
   313→  return Math.max(count, max) + 1;
   314→}

   316→// The event the calendar itself books this Saturday: the inaugural card is
   317→// Fight Night 1; after that, the month's numbered (PPV) slot belongs to the

   320→export function autoEventType(dateISO: string): "numbered" | "fightnight" {
   321→  if (state.events.length === 0) return "fightnight";
   322→  return numberedAvailableThisMonth(dateISO) ? "numbered" : "fightnight";

   331→// THE ROAD AHEAD — the infinite schedule: the next Saturdays and the events
   332→// the calendar would auto-book on them (override any week at creation).
   333→export function upcomingWeeks(count = 6): WeekSlot[] {
   334→  const slots: WeekSlot[] = [];
   335→  let d = currentSaturday();
   336→  let ufc = nextEventNumber();
   337→  let fn = nextFightNightNumber();
   338→  const numberedMonths = new Set(
   339→    state.events.filter((e) => e.type === "numbered").map((e) => e.dateISO.slice(0, 7))
   340→  );
   341→  for (let i = 0; i < count; i++) {
   342→    const iso = isoOf(d);
   343→    const ym = iso.slice(0, 7);
   344→    const type: "numbered" | "fightnight" =
   345→      i === 0 ? autoEventType(iso) : numberedMonths.has(ym) ? "fightnight" : "numbered";
   346→    slots.push({ iso, type, name: type === "numbered" ? `UFC ${ufc}` : `UFC Fight Night ${fn}` });
   347→    if (type === "numbered") {
   348→      ufc++;
   349→      numberedMonths.add(ym);
   350→    } else {
   351→      fn++;
   352→    }
   353→    d = new Date(d.getFullYear(), d.getMonth(), d.getDate() + 7);
   354→  }
   355→  return slots;
   356→}

   376→export function createEvent(type?: "numbered" | "fightnight"): CreateEventResult {
   377→  const iso = isoOf(currentSaturday());
   378→  const chosen = type ?? autoEventType(iso);
   379→  if (state.events.some((e) => e.dateISO === iso)) {
   380→    return { ok: false, reason: "This week already has an event — run it, scrap it, or check the card." };
   381→  }
   382→  if (chosen === "numbered" && !numberedAvailableThisMonth(iso)) {
   383→    return { ok: false, reason: "A numbered event already ran this month — book a Fight Night." };
   384→  }
   385→  const spot = VENUES[state.events.length % VENUES.length];
   386→  const ev: SimEvent = {
   387→    id: uid(),
   388→    type: chosen,
   389→    number: chosen === "numbered" ? nextEventNumber() : nextFightNightNumber(),
   390→    dateISO: iso,
   391→    venue: spot.venue.trim(),
   392→    city: spot.city,
   393→    bouts: [],
   394→    completed: false,
   395→    createdAt: Date.now(),
   396→  };
   397→  state.events = [...state.events, ev];
   398→  save();
   399→  bump();
   400→  return { ok: true, event: ev };
   401→}

   403→export function addBout(eventId: string, aId: string, bId: string, section: EventSection): SimEventBout | null {
   404→  const ev = state.events.find((e) => e.id === eventId);
   405→  const a = getFighter(aId);
   406→  const b = getFighter(bId);
   407→  if (!ev || !a || !b || a.id === b.id) return null;
   408→  // same fighter can't be booked twice on one card
   409→  if (ev.bouts.some((x) => x.aId === a.id || x.bId === a.id || x.aId === b.id || x.bId === b.id)) return null;
   410→  const order = ev.bouts.filter((x) => x.section === section).length;
   411→  const bout: SimEventBout = {
   412→    id: uid(),
   413→    aId: a.id,
   414→    bId: b.id,
   415→    section,
   416→    order,
   417→    rounds: isTitleBout(a, b) ? 5 : 3,
   418→    title: isTitleBout(a, b) || undefined,
   419→    result: null,
   420→  };
   421→  ev.bouts = [...ev.bouts, bout];
   422→  save();
   423→  bump();
   424→  return bout;
   425→}

   427→export function removeBout(eventId: string, boutId: string): void {
   428→  const ev = state.events.find((e) => e.id === eventId);
   429→  if (!ev) return;
   430→  const bout = ev.bouts.find((b) => b.id === boutId);
   431→  if (bout?.result) return; // already fought — it's history now
   432→  ev.bouts = ev.bouts.filter((b) => b.id !== boutId);
   433→  ev.bouts.filter((b) => b.section === bout?.section).forEach((b, i) => (b.order = i));
   434→  save();
   435→  bump();
   436→}

   438→export function moveBout(eventId: string, boutId: string, dir: -1 | 1): void {
   439→  const ev = state.events.find((e) => e.id === eventId);
   440→  if (!ev) return;
   441→  const bout = ev.bouts.find((b) => b.id === boutId);
   442→  if (!bout || bout.result) return;
   443→  const list = ev.bouts.filter((b) => b.section === bout.section).sort((a, b) => a.order - b.order);
   444→  const idx = list.findIndex((b) => b.id === boutId);
   445→  const swapIdx = idx + dir;
   446→  if (swapIdx < 0 || swapIdx >= list.length) return;
   447→  const other = list[swapIdx];
   448→  if (other.result) return;
   449→  const t = bout.order;
   450→  bout.order = other.order;
   451→  other.order = t;
   452→  save();
   453→  bump();
   454→}

   462→// THE moment a universe event bout is decided: careers update, belts can
   463→// change hands (one in a title fight, both in a superfight), and the ledger
   464→// grows. Lab fights NEVER call this. Doubly guarded: the bout's own result
   465→// flag AND the ledger's boutId key — a fight can only ever be written once.
   466→export function recordEventBout(eventId: string, boutId: string, sim: SimResult): RecordOutcome | null {
   467→  const ev = state.events.find((e) => e.id === eventId);
   468→  const bout = ev?.bouts.find((b) => b.id === boutId);
   469→  if (!ev || !bout || bout.result) return null;
   470→  if (state.ledger.some((r) => r.boutId === bout.id)) return null;
   471→  const a = getFighter(bout.aId);
   472→  const b = getFighter(bout.bId);
   473→  if (!a || !b) return null;
   474→
   475→  // championship status is computed at the bell — belts can already have
   476→  // moved earlier on this same card
   477→  const title = bout.title || isTitleBout(a, b);
   478→  const date = prettyDate(ev.dateISO);
   479→  const rec: UniverseBout = {
   480→    boutId: bout.id,
   481→    aId: a.id,
   482→    bId: b.id,
   483→    winner: sim.winner,
   484→    method: sim.method,
   485→    detail: sim.detail,
   486→    round: sim.round,
   487→    time: sim.time,
   488→    evName: eventDisplayName(ev),
   489→    section: bout.section,
   490→    date,
   491→    wc: boutWeightClass(a, b),
   492→    title,
   493→  };
   494→  applyBout(a, b, rec);
   495→  state.ledger = [...state.ledger, rec];
   496→
   497→  bout.result = {
   498→    winner: sim.winner,
   499→    method: sim.method,
   500→    detail: sim.detail,
   501→    round: sim.round,
   502→    time: sim.time,
   503→  };
   504→  if (title && sim.winner !== "A" && sim.winner !== "B") bout.title = undefined;
   505→
   506→  // belts can change hands right here — one, or every belt in a superfight
   507→  const belts = transferBelts(a, b, sim.winner, rec.wc);
   508→
   509→  if (ev.bouts.every((x) => x.result)) ev.completed = true;
   510→  save();
   511→  bump();
   512→
   513→  const w = sim.winner === "A" ? a : b;
   514→  const l = sim.winner === "A" ? b : a;
   515→  return {
   516→    evName: eventDisplayName(ev),
   517→    belts,
   518→    note:
   519→      sim.winner === "DRAW" || sim.winner === "NC"
   520→        ? `${a.name} vs ${b.name} — a draw goes on both records at ${eventDisplayName(ev)}`
   521→        : `${w.name} def. ${l.name} — written onto both career records`,
   522→  };
   523→}

   525→// ---------- OCTAGONHQ rankings ----------
   526→// Points come ONLY from universe event fights: decision win +3, finish +4,
   527→// title-fight win +1, draw +1. Champion sits on top; ties break on wins,
   528→// finishes, then real-world seeding.
   529→export interface Standing {
   530→  fighter: Fighter;
   531→  pts: number;
   532→  simW: number;
   533→  simL: number;
   534→  simD: number;
   535→  finishes: number;
   536→  rank: number | "C";
   537→}
   538→
   539→function pointsFromLedger(): Map<string, { pts: number; finishes: number }> {
   540→  const map = new Map<string, { pts: number; finishes: number }>();
   541→  for (const b of state.ledger) {
   542→    const winnerId = b.winner === "A" ? b.aId : b.winner === "B" ? b.bId : null;
   543→    const finish = b.method === "KO" || b.method === "SUB";
   544→    if (winnerId) {
   545→      const cur = map.get(winnerId) ?? { pts: 0, finishes: 0 };
   546→      cur.pts += (finish ? 4 : 3) + (b.title ? 1 : 0);
   547→      if (finish) cur.finishes += 1;
   548→      map.set(winnerId, cur);
   549→    } else {
   550→      for (const id of [b.aId, b.bId]) {
   551→        const cur = map.get(id) ?? { pts: 0, finishes: 0 };
   552→        cur.pts += 1;
   553→        map.set(id, cur);
   554→      }
   555→    }
   556→  }
   557→  return map;
   558→}
   559→
   560→export function divisionStandings(div: DivisionId): Standing[] {
   561→  const pts = pointsFromLedger();
   562→  // the division's champ can live anywhere on the roster — a DOUBLE CHAMP
   563→  // (superfight winner) holds this belt even if he fights out of another
   564→  // division, so search the full roster, not just this division's list
   565→  const champ = ROSTER.find((f) => champDivisions(f).includes(div));
   566→  const others = getDivision(div)
   567→    .filter((f) => !champDivisions(f).includes(div))
   568→    .map((f) => {
   569→      const p = pts.get(f.id) ?? { pts: 0, finishes: 0 };
   570→      const s = simRecord(f);
   571→      return { fighter: f, pts: p.pts, finishes: p.finishes, simW: s.w, simL: s.l, simD: s.d, rank: 0 as number | "C" };
   572→    })
   573→    .sort((x, y) => {
   574→      if (y.pts !== x.pts) return y.pts - x.pts;
   575→      if (y.simW !== x.simW) return y.simW - x.simW;
   576→      if (y.finishes !== x.finishes) return y.finishes - x.finishes;
   577→      const rx = x.fighter.rank ?? 99;
   578→      const ry = y.fighter.rank ?? 99;
   579→      if (rx !== ry) return rx - ry;
   580→      return y.fighter.elo - x.fighter.elo;
   581→    });
   582→  const out: Standing[] = [];
   583→  if (champ) {
   584→    const p = pts.get(champ.id) ?? { pts: 0, finishes: 0 };
   585→    const s = simRecord(champ);
   586→    out.push({ fighter: champ, pts: p.pts, finishes: p.finishes, simW: s.w, simL: s.l, simD: s.d, rank: "C" });
   587→  }
   588→  others.forEach((o, i) => {
   589→    o.rank = i + 1;
   590→    out.push(o as Standing);
   591→  });
   592→  return out;
   593→}

   595→// OCTAGONHQ rank label for a fighter: "C" (champ), "#1".."#15", or "UNRANKED".
   596→// Mirrors DivisionsTab — only the top 15 contenders are ranked.
   597→// Cached per universe version so bulk consumers (picker sort) stay instant.
   598→let rankLabelCache: { version: number; map: Map<string, string> } | null = null;
   599→
   600→export function hqRankMap(): Map<string, string> {
   601→  if (rankLabelCache && rankLabelCache.version === version) return rankLabelCache.map;
   602→  const map = new Map<string, string>();
   603→  for (const d of Object.keys(DIVISION_MAP) as DivisionId[]) {
   604→    for (const s of divisionStandings(d)) {
   605→      map.set(s.fighter.id, s.rank === "C" ? "C" : s.rank <= 15 ? `#${s.rank}` : "UNRANKED");
   606→    }
   607→  }
   608→  rankLabelCache = { version, map };
   609→  return map;
   610→}
   611→
   612→export function hqRankLabel(fighterId: string): string {
   613→  const f = ROSTER.find((x) => x.id === fighterId);
   614→  if (!f) return "UNRANKED";
   615→  if (f.isChampion) return "C";
   616→  return hqRankMap().get(f.id) ?? "UNRANKED";
   617→}

   624→// universe pound-for-pound: champs by points, then top points scorers
   625→export function simP4P(gender: "men" | "women"): { fighter: Fighter; pts: number; rank: number | "C" }[] {
   626→  const pts = pointsFromLedger();
   627→  const pool = ROSTER.filter((f) => DIVISION_MAP[f.division].division === gender);
   628→  const champs = pool.filter((f) => champDivisions(f).length > 0);
   629→  const rest = pool.filter((f) => !f.isChampion);
   630→  const champsRanked = champs.map((f) => ({ fighter: f, pts: pts.get(f.id)?.pts ?? 0, rank: "C" as number | "C" }));
   631→  const restRanked = rest
   632→    .map((f) => ({ fighter: f, pts: pts.get(f.id)?.pts ?? 0, rank: 0 as number | "C" }))
   633→    .sort((a, b) => b.pts - a.pts || b.fighter.elo - a.fighter.elo)
   634→    .slice(0, Math.max(0, 10 - champs.length));
   635→  return [...champsRanked, ...restRanked].slice(0, 10);
   636→}

   638→// ---------- persistence ----------
   639→export function rehydrateUniverse(): void {
   640→  if (hydrated || typeof window === "undefined") return;
   641→  hydrated = true;
   642→  // wipe the old fight-lab careers cache — those records were never real
   643→  try {
   644→    window.localStorage.removeItem(LEGACY_CAREERS_KEY);
   645→  } catch {
   646→    /* ignore */
   647→  }
   648→  let raw: string | null = null;
   649→  try {
   650→    raw = window.localStorage.getItem(STORAGE_KEY);
   651→  } catch {
   652→    raw = null;
   653→  }
   654→  if (!raw) return;
   655→  try {
   656→    const parsed = JSON.parse(raw) as UniverseState;
   657→    if (!parsed || !Array.isArray(parsed.events)) return;
   658→    state = { events: parsed.events, ledger: [] };
   659→
   660→    // migrate: legacy fight nights get their infinite numbers (Fight Night
   661→    // 1, 2, 3… in fight order); the old display name is kept so any open
   662→    // wallet bets placed under it still settle
   663→    let fn = 1;
   664→    let changed = false;
   665→    const ordered = [...state.events].sort((a, b) => (a.dateISO < b.dateISO ? -1 : a.dateISO > b.dateISO ? 1 : (a.createdAt ?? 0) - (b.createdAt ?? 0)));
   666→    for (const ev of ordered) {
   667→      if (ev.type !== "fightnight") continue;
   668→      if (ev.number == null) {
   669→        ev.legacyName = eventDisplayName(ev); // capture BEFORE numbering
   670→        ev.number = fn;
   671→        changed = true;
   672→      }
   673→      fn = Math.max(fn, ev.number + 1);
   674→    }
   675→
   676→    // REBUILD THE LEDGER FROM EVENT RESULTS — the stored ledger array is
   677→    // deliberately IGNORED. Older builds could append the same bout several
   678→    // times (the 4x duplication bug: every fight showed as four identical
   679→    // history entries and records counted 4x). Events are the source of
   680→    // truth: every fought bout carries exactly ONE result object, so
   681→    // rebuilding here heals any duplicated ledger on load — records,
   682→    // histories, rankings and belt lineage all come out exact.
   683→    for (const ev of ordered) {
   684→      for (const bout of [...ev.bouts].sort((x, y) => x.order - y.order)) {
   685→        if (!bout.result) continue;
   686→        const a = getFighter(bout.aId);
   687→        const b = getFighter(bout.bId);
   688→        if (!a || !b || a.id === b.id) continue;
   689→        const rec: UniverseBout = {
   690→          boutId: bout.id,
   691→          aId: a.id,
   692→          bId: b.id,
   693→          winner: bout.result.winner,
   694→          method: bout.result.method,
   695→          detail: bout.result.detail,
   696→          round: bout.result.round,
   697→          time: bout.result.time,
   698→          evName: eventDisplayName(ev),
   699→          section: bout.section,
   700→          date: prettyDate(ev.dateISO),
   701→          wc: boutWeightClass(a, b),
   702→          title: !!bout.title,
   703→        };
   704→        applyBout(a, b, rec);
   705→        transferBelts(a, b, rec.winner, rec.wc);
   706→        state.ledger.push(rec);
   707→      }
   708→      if (!ev.completed && ev.bouts.length > 0 && ev.bouts.every((x) => x.result)) {
   709→        ev.completed = true;
   710→        changed = true;
   711→      }
   712→    }
   713→    // persist the healed universe (deduped ledger + boutId keys)
   714→    save();
   715→    if (changed || state.ledger.length > 0) bump();
   716→  } catch {
   717→    state = { events: [], ledger: [] };
   718→  }
   719→}

   721→// wipe the universe and reload back to real Oct 17, 2026 data
   722→export function resetUniverse(): void {
   723→  state = { events: [], ledger: [] };
   724→  save();
   725→  if (typeof window !== "undefined") window.location.reload();
   726→}
