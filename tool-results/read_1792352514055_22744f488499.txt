    46→function methodText(res: NonNullable<SimEventBout["result"]>): string {
    47→  const base =
    48→    res.method === "KO" ? `KO/TKO${res.detail ? ` (${res.detail})` : ""}` :
    49→    res.method === "SUB" ? `SUB${res.detail ? ` (${res.detail})` : ""}` :
    50→    res.method === "UDEC" ? "Unanimous Dec" :
    51→    res.method === "SDEC" ? "Split Dec" :
    52→    res.method === "MDEC" ? "Majority Dec" : "DQ";
    53→  return res.method === "KO" || res.method === "SUB" ? `${base} · R${res.round} ${res.time}` : base;
    54→}

    56→// settle any open wallet bets that target this pair under this event label
    57→function settleBetsForBout(aId: string, bId: string, label: string, sim: SimResult) {
    58→  const a = getFighter(aId);
    59→  const b = getFighter(bId);
    60→  if (!a || !b) return { won: 0, lost: 0, staked: 0 };
    61→  const note = resultSummary(sim, a, b);
    62→  const winnerId = sim.winner === "A" ? aId : sim.winner === "B" ? bId : "";
    63→  let won = 0;
    64→  let lost = 0;
    65→  let staked = 0;
    66→  for (const bet of useBetStore.getState().bets) {
    67→    if (bet.status !== "open" || bet.eventLabel !== label) continue;
    68→    if (
    69→      !((bet.bettorFighterId === aId && bet.opponentId === bId) ||
    70→        (bet.bettorFighterId === bId && bet.opponentId === aId))
    71→    ) continue;
    72→    const isWin = bet.bettorFighterId === winnerId;
    73→    const payout = bet.stake * (bet.decOdds - 1);
    74→    useBetStore.getState().settleBet(bet.id, isWin, note, payout);
    75→    if (isWin) {
    76→      won++;
    77→    } else {
    78→      lost++;
    79→      staked += bet.stake;
    80→    }
    81→  }
    82→  return { won, lost, staked };
    83→}

   421→function EventCard({ ev }: { ev: SimEvent }) {
   422→  const { setSimSpec } = useApp();
   423→  const { bets } = useBetStore();
   424→  const name = eventDisplayName(ev);
   425→  const pending = ev.bouts.filter((b) => !b.result);
   426→  const openBetsCount = bets.filter(
   427→    (b) => b.status === "open" && b.eventLabel === name
   428→  ).length;
   429→
   430→  const runFullCard = () => {
   431→    if (pending.length === 0) return;
   432→    // broadcast order: early prelims → prelims → main card
   433→    const order: EventSection[] = ["early", "prelim", "main"];
   434→    let won = 0;
   435→    let lost = 0;
   436→    let staked = 0;
   437→    let belts = 0;
   438→    for (const section of order) {
   439→      for (const bout of eventBouts(ev, section)) {
   440→        if (bout.result) continue;
   441→        const a = getFighter(bout.aId)!;
   442→        const b = getFighter(bout.bId)!;
   443→        if (!a || !b) continue;
   444→        const sim = simulateFight(a, b, bout.rounds);
   445→        const outcome = recordEventBout(ev.id, bout.id, sim);
   446→        if (outcome) {
   447→          if (outcome.belt) {
   448→            belts++;
   449→            toast.success(
   450→              `NEW CHAMPION! ${outcome.belt.newChamp.name} dethrones ${outcome.belt.oldChamp.name} — the ${outcome.belt.divName} belt changes hands!`,
   451→              { duration: 7000 }
   452→            );
   453→          }
   454→          const s = settleBetsForBout(a.id, b.id, name, sim);
   455→          won += s.won;
   456→          lost += s.lost;
   457→          staked += s.staked;
   458→        }
   459→      }
   460→    }
   461→    toast.success(`${name} is in the books — records, rankings & titles updated.`, { duration: 5000 });
   462→    if (won + lost > 0) {
   463→      if (won > 0) toast.success(`Event bets: ${won}W-${lost}L`);
   464→      else toast.error(`Event bets: 0W-${lost}L · −$${staked.toFixed(2)}`);
   465→    }
   466→  };
   467→
   468→  const mainBout = eventBouts(ev, "main")[0];
   469→
   470→  return (
   471→    <motion.section
   472→      initial={{ opacity: 0, y: 14 }}
   473→      animate={{ opacity: 1, y: 0 }}
   474→      className="overflow-hidden rounded-2xl border border-white/10 bg-white/[0.015]"
   475→    >
   476→      <div className="relative border-b border-white/10 px-4 py-4 sm:px-5">

   568→// ---------- past event row ----------
   569→function PastEventRow({ ev }: { ev: SimEvent }) {
   570→  const [open, setOpen] = useState(false);
   571→  const name = eventDisplayName(ev);
   572→  const mainBout = eventBouts(ev, "main")[0];
   573→  const mainA = mainBout ? getFighter(mainBout.aId) : null;
   574→  const mainB = mainBout ? getFighter(mainBout.bId) : null;
   575→  const mainWin = mainBout?.result
   576→    ? mainBout.result.winner === "A" ? mainA : mainBout.result.winner === "B" ? mainB : null
   577→    : null;
   578→
   579→  return (
   580→    <div className="overflow-hidden rounded-xl border border-white/10 bg-white/[0.015]">
   581→      <button onClick={() => setOpen(!open)} className="flex w-full items-center gap-3 px-3.5 py-3 text-left transition-colors hover:bg-white/[0.03]">
   582→        <div className="min-w-0 flex-1">
   583→          <div className="flex items-center gap-2">
   584→            <span className={`text-[9px] font-display font-bold tracking-widest ${ev.type === "numbered" ? "text-[#e0b64c]" : "text-sky-400"}`}>
   585→              {ev.type === "numbered" ? `UFC ${ev.number}` : "FIGHT NIGHT"}
   586→            </span>
   587→            <span className="num text-[10px] text-zinc-600">{prettyDate(ev.dateISO)}</span>
   588→          </div>
   589→          <div className="truncate font-display text-sm font-bold text-white">{name}</div>
   590→          {mainWin && mainA && mainB && (
   591→            <div className="truncate text-[10px] text-zinc-500">
   592→              {mainWin.name} def. {mainWin === mainA ? mainB.name : mainA.name}
   593→              {mainBout?.result && ` · ${methodText(mainBout.result)}`}
   594→            </div>
   595→          )}
   596→        </div>
   597→        <Trophy className="h-4 w-4 shrink-0 text-zinc-600" />
   598→        <ChevronDown className={`h-4 w-4 shrink-0 text-zinc-600 transition-transform ${open ? "" : "-rotate-90"}`} />
   599→      </button>
   600→      {open && (
   601→        <div className="space-y-1.5 border-t border-white/5 px-3.5 py-3">
   602→          {(["early", "prelim", "main"] as EventSection[]).flatMap((section) =>
   603→            eventBouts(ev, section).map((bout) => {
   604→              const a = getFighter(bout.aId);
   605→              const b = getFighter(bout.bId);
   606→              if (!a || !b || !bout.result) return null;
   607→              const w = bout.result.winner === "A" ? a : bout.result.winner === "B" ? b : null;
   608→              return (
   609→                <div key={bout.id} className="flex items-center justify-between gap-2 rounded-lg bg-white/[0.02] px-2.5 py-1.5 text-[11px]">
   610→                  <span className="shrink-0 text-[8px] font-display tracking-widest text-zinc-600">
   611→                    {boutSectionLabel(ev, bout)}
   612→                  </span>
   613→                  <span className="min-w-0 flex-1 truncate text-zinc-400">
   614→                    <span className="font-semibold text-zinc-200">{w ? w.name : "Draw"}</span>
   615→                    {w ? ` def. ${w === a ? b.name : a.name}` : ` — ${a.name} vs ${b.name}`}
   616→                  </span>
   617→                  <span className="shrink-0 text-[10px] text-zinc-500">{methodText(bout.result)}</span>
   618→                </div>
   619→              );
   620→            })
   621→          )}
   622→        </div>
   623→      )}
   624→    </div>
   625→  );
   626→}
