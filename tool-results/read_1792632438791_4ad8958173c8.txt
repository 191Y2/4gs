    61→    61→  const latest = state.events.reduce((m, e) => (e.dateISO > m.dateISO ? e : m));
    62→    62→  if (!latest.completed) return dateOfIso(latest.dateISO);
    63→    63→  // card done — the next fight week unlocks instantly, forever
    64→    64→  const next = dateOfIso(latest.dateISO);
    65→    65→  next.setDate(next.getDate() + 7);
    66→    66→  return next;
    67→    67→}

   133→   133→function boutWeightClass(a: Fighter, b: Fighter): DivisionId {
   134→   134→  if (a.division === b.division) return a.division;
   135→   135→  // cross-division BELT DEFENSE: a fighter holding a belt in the other's
   136→   136→  // division fights there — a double/triple champ booked against a
   137→   137→  // division's contender defends that belt at that weight
   138→   138→  const beltsA = champDivisions(a);
   139→   139→  const beltsB = champDivisions(b);
   140→   140→  const aDefends = beltsA.includes(b.division);
   141→   141→  const bDefends = beltsB.includes(a.division);
   142→   142→  if (aDefends && !bDefends) return b.division;
   143→   143→  if (bDefends && !aDefends) return a.division;
   144→   144→  // superfight / plain cross-division booking: heavier weight class
   145→   145→  return (DIVISION_MAP[a.division]?.limit ?? 0) >= (DIVISION_MAP[b.division]?.limit ?? 0)
   146→   146→    ? a.division
   147→   147→    : b.division;
   148→   148→}

   150→   150→// same-division bout with a reigning champion = classic title fight. A
   151→   151→// champ vs champ cross-division bout = SUPERFIGHT — all belts on the line
   152→   152→// and the winner becomes a double/triple/…/8-division champ. A fighter
   153→   153→// holding a belt in the OTHER fighter's division = cross-division belt
   154→   154→// defense — also a title fight. Everyone in this universe fights in their
   155→   155→// prime; a legend booked against the champ is a title fight and the belt
   156→   156→// CAN move, same as anyone else.
   157→   157→export function isTitleBout(a: Fighter, b: Fighter): boolean {
   158→   158→  if (isSuperfight(a, b)) return true;
   159→   159→  const beltsA = champDivisions(a);
   160→   160→  const beltsB = champDivisions(b);
   161→   161→  if (beltsA.includes(b.division) || beltsB.includes(a.division)) return true;
   162→   162→  if (a.division === b.division) return !!(a.isChampion || b.isChampion);
   163→   163→  return false;
   164→   164→}

   211→   211→  else if (resB === "L") b.simL = (b.simL ?? 0) + 1;
   212→   212→  else if (resB === "D") b.simD = (b.simD ?? 0) + 1;

   331→   331→// THE ROAD AHEAD — the infinite schedule: the next Saturdays and the events
   332→   332→// the calendar would auto-book on them (override any week at creation).
   333→   333→export function upcomingWeeks(count = 6): WeekSlot[] {
   334→   334→  const slots: WeekSlot[] = [];
   335→   335→  let d = currentSaturday();
   336→   336→  let ufc = nextEventNumber();
   337→   337→  let fn = nextFightNightNumber();
   338→   338→  const numberedMonths = new Set(
   339→   339→    state.events.filter((e) => e.type === "numbered").map((e) => e.dateISO.slice(0, 7))
   340→   340→  );
   341→   341→  for (let i = 0; i < count; i++) {
   342→   342→    const iso = isoOf(d);
   343→   343→    const ym = iso.slice(0, 7);
   344→   344→    const type: "numbered" | "fightnight" =
   345→   345→      i === 0 ? autoEventType(iso) : numberedMonths.has(ym) ? "fightnight" : "numbered";
   346→   346→    slots.push({ iso, type, name: type === "numbered" ? `UFC ${ufc}` : `UFC Fight Night ${fn}` });
   347→   347→    if (type === "numbered") {
   348→   348→      ufc++;
   349→   349→      numberedMonths.add(ym);
   350→   350→    } else {
   351→   351→      fn++;
   352→   352→    }
   353→   353→    d = new Date(d.getFullYear(), d.getMonth(), d.getDate() + 7);
   354→   354→  }
   355→   355→  return slots;
   356→   356→}

   376→   376→export function createEvent(type?: "numbered" | "fightnight"): CreateEventResult {
   377→   377→  const iso = isoOf(currentSaturday());
   378→   378→  const chosen = type ?? autoEventType(iso);
   379→   379→  if (state.events.some((e) => e.dateISO === iso)) {
   380→   380→    return { ok: false, reason: "This week already has an event — run it, scrap it, or check the card." };
   381→   381→  }
   382→   382→  if (chosen === "numbered" && !numberedAvailableThisMonth(iso)) {
   383→   383→    return { ok: false, reason: "A numbered event already ran this month — book a Fight Night." };
   384→   384→  }
   385→   385→  const spot = VENUES[state.events.length % VENUES.length];
   386→   386→  const ev: SimEvent = {
   387→   387→    id: uid(),
   388→   388→    type: chosen,
   389→   389→    number: chosen === "numbered" ? nextEventNumber() : nextFightNightNumber(),
   390→   390→    dateISO: iso,
   391→   391→    venue: spot.venue.trim(),
   392→   392→    city: spot.city,
   393→   393→    bouts: [],
   394→   394→    completed: false,
   395→   395→    createdAt: Date.now(),
   396→   396→  };
   397→   397→  state.events = [...state.events, ev];
   398→   398→  save();
   399→   399→  bump();
   400→   400→  return { ok: true, event: ev };
   401→   401→}

   403→   403→export function addBout(eventId: string, aId: string, bId: string, section: EventSection): SimEventBout | null {
   404→   404→  const ev = state.events.find((e) => e.id === eventId);
   405→   405→  const a = getFighter(aId);
   406→   406→  const b = getFighter(bId);
   407→   407→  if (!ev || !a || !b || a.id === b.id) return null;
   408→   408→  // same fighter can't be booked twice on one card
   409→   409→  if (ev.bouts.some((x) => x.aId === a.id || x.bId === a.id || x.aId === b.id || x.bId === b.id)) return null;
   410→   410→  const order = ev.bouts.filter((x) => x.section === section).length;
   411→   411→  const bout: SimEventBout = {
   412→   412→    id: uid(),
   413→   413→    aId: a.id,
   414→   414→    bId: b.id,
   415→   415→    section,
   416→   416→    order,
   417→   417→    rounds: isTitleBout(a, b) ? 5 : 3,
   418→   418→    title: isTitleBout(a, b) || undefined,
   419→   419→    result: null,
   420→   420→  };
   421→   421→  ev.bouts = [...ev.bouts, bout];
   422→   422→  save();
   423→   423→  bump();
   424→   424→  return bout;
   425→   425→}

   427→   427→export function removeBout(eventId: string, boutId: string): void {
   428→   428→  const ev = state.events.find((e) => e.id === eventId);
   429→   429→  if (!ev) return;
   430→   430→  const bout = ev.bouts.find((b) => b.id === boutId);
   431→   431→  if (bout?.result) return; // already fought — it's history now
   432→   432→  ev.bouts = ev.bouts.filter((b) => b.id !== boutId);
   433→   433→  ev.bouts.filter((b) => b.section === bout?.section).forEach((b, i) => (b.order = i));
   434→   434→  save();
   435→   435→  bump();
   436→   436→}

   438→   438→export function moveBout(eventId: string, boutId: string, dir: -1 | 1): void {
   439→   439→  const ev = state.events.find((e) => e.id === eventId);
   440→   440→  if (!ev) return;
   441→   441→  const bout = ev.bouts.find((b) => b.id === boutId);
   442→   442→  if (!bout || bout.result) return;
   443→   443→  const list = ev.bouts.filter((b) => b.section === bout.section).sort((a, b) => a.order - b.order);
   444→   444→  const idx = list.findIndex((b) => b.id === boutId);
   445→   445→  const swapIdx = idx + dir;
   446→   446→  if (swapIdx < 0 || swapIdx >= list.length) return;
   447→   447→  const other = list[swapIdx];
   448→   448→  if (other.result) return;
   449→   449→  const t = bout.order;
   450→   450→  bout.order = other.order;
   451→   451→  other.order = t;
   452→   452→  save();
   453→   453→  bump();
   454→   454→}

   462→   462→// THE moment a universe event bout is decided: careers update, belts can
   463→   463→// change hands (one in a title fight, both in a superfight), and the ledger
   464→   464→// grows. Lab fights NEVER call this. Doubly guarded: the bout's own result
   465→   465→// flag AND the ledger's boutId key — a fight can only ever be written once.
   466→   466→export function recordEventBout(eventId: string, boutId: string, sim: SimResult): RecordOutcome | null {
   467→   467→  const ev = state.events.find((e) => e.id === eventId);
   468→   468→  const bout = ev?.bouts.find((b) => b.id === boutId);
   469→   469→  if (!ev || !bout || bout.result) return null;
   470→   470→  if (state.ledger.some((r) => r.boutId === bout.id)) return null;
   471→   471→  const a = getFighter(bout.aId);
   472→   472→  const b = getFighter(bout.bId);
   473→   473→  if (!a || !b) return null;
   474→   474→
   475→   475→  // championship status is computed at the bell — belts can already have
   476→   476→  // moved earlier on this same card
   477→   477→  const title = bout.title || isTitleBout(a, b);
   478→   478→  const date = prettyDate(ev.dateISO);
   479→   479→  const rec: UniverseBout = {
   480→   480→    boutId: bout.id,
   481→   481→    aId: a.id,
   482→   482→    bId: b.id,
   483→   483→    winner: sim.winner,
   484→   484→    method: sim.method,
   485→   485→    detail: sim.detail,
   486→   486→    round: sim.round,
   487→   487→    time: sim.time,
   488→   488→    evName: eventDisplayName(ev),
   489→   489→    section: bout.section,
   490→   490→    date,
   491→   491→    wc: boutWeightClass(a, b),
   492→   492→    title,
   493→   493→  };
   494→   494→  applyBout(a, b, rec);
   495→   495→  state.ledger = [...state.ledger, rec];
   496→   496→
   497→   497→  bout.result = {
   498→   498→    winner: sim.winner,
   499→   499→    method: sim.method,
   500→   500→    detail: sim.detail,
   501→   501→    round: sim.round,
   502→   502→    time: sim.time,
   503→   503→  };
   504→   504→  if (title && sim.winner !== "A" && sim.winner !== "B") bout.title = undefined;
   505→   505→
   506→   506→  // belts can change hands right here — one, or every belt in a superfight
   507→   507→  const belts = transferBelts(a, b, sim.winner, rec.wc);
   508→   508→
   509→   509→  if (ev.bouts.every((x) => x.result)) ev.completed = true;
   510→   510→  save();
   511→   511→  bump();
   512→   512→
   513→   513→  const w = sim.winner === "A" ? a : b;
   514→   514→  const l = sim.winner === "A" ? b : a;
   515→   515→  return {
   516→   516→    evName: eventDisplayName(ev),
   517→   517→    belts,
   518→   518→    note:
   519→   519→      sim.winner === "DRAW" || sim.winner === "NC"
   520→   520→        ? `${a.name} vs ${b.name} — a draw goes on both records at ${eventDisplayName(ev)}`
   521→   521→        : `${w.name} def. ${l.name} — written onto both career records`,
   522→   522→  };
   523→   523→}

   525→   525→// ---------- OCTAGONHQ rankings ----------
   526→   526→// Points come ONLY from universe event fights: decision win +3, finish +4,
   527→   527→// title-fight win +1, draw +1. Champion sits on top; ties break on wins,
   528→   528→// finishes, then real-world seeding.
   529→   529→export interface Standing {
   530→   530→  fighter: Fighter;
   531→   531→  pts: number;
   532→   532→  simW: number;
   533→   533→  simL: number;
   534→   534→  simD: number;
   535→   535→  finishes: number;
   536→   536→  rank: number | "C";
   537→   537→}
   538→   538→
   539→   539→function pointsFromLedger(): Map<string, { pts: number; finishes: number }> {
   540→   540→  const map = new Map<string, { pts: number; finishes: number }>();
   541→   541→  for (const b of state.ledger) {
   542→   542→    const winnerId = b.winner === "A" ? b.aId : b.winner === "B" ? b.bId : null;
   543→   543→    const finish = b.method === "KO" || b.method === "SUB";
   544→   544→    if (winnerId) {
   545→   545→      const cur = map.get(winnerId) ?? { pts: 0, finishes: 0 };
   546→   546→      cur.pts += (finish ? 4 : 3) + (b.title ? 1 : 0);
   547→   547→      if (finish) cur.finishes += 1;
   548→   548→      map.set(winnerId, cur);
   549→   549→    } else {
   550→   550→      for (const id of [b.aId, b.bId]) {
   551→   551→        const cur = map.get(id) ?? { pts: 0, finishes: 0 };
   552→   552→        cur.pts += 1;
   553→   553→        map.set(id, cur);
   554→   554→      }
   555→   555→    }
   556→   556→  }
   557→   557→  return map;
   558→   558→}
   559→   559→
   560→   560→export function divisionStandings(div: DivisionId): Standing[] {
   561→   561→  const pts = pointsFromLedger();
   562→   562→  // the division's champ can live anywhere on the roster — a DOUBLE CHAMP
   563→   563→  // (superfight winner) holds this belt even if he fights out of another
   564→   564→  // division, so search the full roster, not just this division's list
   565→   565→  const champ = ROSTER.find((f) => champDivisions(f).includes(div));
   566→   566→  const others = getDivision(div)
   567→   567→    .filter((f) => !champDivisions(f).includes(div))
   568→   568→    .map((f) => {
   569→   569→      const p = pts.get(f.id) ?? { pts: 0, finishes: 0 };
   570→   570→      const s = simRecord(f);
   571→   571→      return { fighter: f, pts: p.pts, finishes: p.finishes, simW: s.w, simL: s.l, simD: s.d, rank: 0 as number | "C" };
   572→   572→    })
   573→   573→    .sort((x, y) => {
   574→   574→      if (y.pts !== x.pts) return y.pts - x.pts;
   575→   575→      if (y.simW !== x.simW) return y.simW - x.simW;
   576→   576→      if (y.finishes !== x.finishes) return y.finishes - x.finishes;
   577→   577→      const rx = x.fighter.rank ?? 99;
   578→   578→      const ry = y.fighter.rank ?? 99;
   579→   579→      if (rx !== ry) return rx - ry;
   580→   580→      return y.fighter.elo - x.fighter.elo;
   581→   581→    });
   582→   582→  const out: Standing[] = [];
   583→   583→  if (champ) {
   584→   584→    const p = pts.get(champ.id) ?? { pts: 0, finishes: 0 };
   585→   585→    const s = simRecord(champ);
   586→   586→    out.push({ fighter: champ, pts: p.pts, finishes: p.finishes, simW: s.w, simL: s.l, simD: s.d, rank: "C" });
   587→   587→  }
   588→   588→  others.forEach((o, i) => {
   589→   589→    o.rank = i + 1;
   590→   590→    out.push(o as Standing);
   591→   591→  });
   592→   592→  return out;
   593→   593→}

   595→   595→// OCTAGONHQ rank label for a fighter: "C" (champ), "#1".."#15", or "UNRANKED".
   596→   596→// Mirrors DivisionsTab — only the top 15 contenders are ranked.
   597→   597→// Cached per universe version so bulk consumers (picker sort) stay instant.
   598→   598→let rankLabelCache: { version: number; map: Map<string, string> } | null = null;
   599→   599→
   600→   600→export function hqRankMap(): Map<string, string> {
   601→   601→  if (rankLabelCache && rankLabelCache.version === version) return rankLabelCache.map;
   602→   602→  const map = new Map<string, string>();
   603→   603→  for (const d of Object.keys(DIVISION_MAP) as DivisionId[]) {
   604→   604→    for (const s of divisionStandings(d)) {
   605→   605→      map.set(s.fighter.id, s.rank === "C" ? "C" : s.rank <= 15 ? `#${s.rank}` : "UNRANKED");
   606→   606→    }
   607→   607→  }
   608→   608→  rankLabelCache = { version, map };
   609→   609→  return map;
   610→   610→}

   624→   624→// universe pound-for-pound: champs by points, then top points scorers
   625→   625→export function simP4P(gender: "men" | "women"): { fighter: Fighter; pts: number; rank: number | "C" }[] {
   626→   626→  const pts = pointsFromLedger();
   627→   627→  const pool = ROSTER.filter((f) => DIVISION_MAP[f.division].division === gender);
   628→   628→  const champs = pool.filter((f) => champDivisions(f).length > 0);
   629→   629→  const rest = pool.filter((f) => !f.isChampion);
   630→   630→  const champsRanked = champs.map((f) => ({ fighter: f, pts: pts.get(f.id)?.pts ?? 0, rank: "C" as number | "C" }));
   631→   631→  const restRanked = rest
   632→   632→    .map((f) => ({ fighter: f, pts: pts.get(f.id)?.pts ?? 0, rank: 0 as number | "C" }))
   633→   633→    .sort((a, b) => b.pts - a.pts || b.fighter.elo - a.fighter.elo)
   634→   634→    .slice(0, Math.max(0, 10 - champs.length));
   635→   635→  return [...champsRanked, ...restRanked].slice(0, 10);
   636→   636→}

   638→   638→// ---------- persistence ----------
   639→   639→export function rehydrateUniverse(): void {
   640→   640→  if (hydrated || typeof window === "undefined") return;
   641→   641→  hydrated = true;
   642→   642→  // wipe the old fight-lab careers cache — those records were never real
   643→   643→  try {
   644→   644→    window.localStorage.removeItem(LEGACY_CAREERS_KEY);
   645→   645→  } catch {
   646→   646→    /* ignore */
   647→   647→  }
   648→   648→  let raw: string | null = null;
   649→   649→  try {
   650→   650→    raw = window.localStorage.getItem(STORAGE_KEY);
   651→   651→  } catch {
   652→   652→    raw = null;
   653→   653→  }
   654→   654→  if (!raw) return;
   655→   655→  try {
   656→   656→    const parsed = JSON.parse(raw) as UniverseState;
   657→   657→    if (!parsed || !Array.isArray(parsed.events)) return;
   658→   658→    state = { events: parsed.events, ledger: [] };
   659→   659→
   660→   660→    // migrate: legacy fight nights get their infinite numbers (Fight Night
   661→   661→    // 1, 2, 3… in fight order); the old display name is kept so any open
   662→   662→    // wallet bets placed under it still settle
   663→   663→    let fn = 1;
   664→   664→    let changed = false;
   665→   665→    const ordered = [...state.events].sort((a, b) => (a.dateISO < b.dateISO ? -1 : a.dateISO > b.dateISO ? 1 : (a.createdAt ?? 0) - (b.createdAt ?? 0)));
   666→   666→    for (const ev of ordered) {
   667→   667→      if (ev.type !== "fightnight") continue;
   668→   668→      if (ev.number == null) {
   669→   669→        ev.legacyName = eventDisplayName(ev); // capture BEFORE numbering
   670→   670→        ev.number = fn;
   671→   671→        changed = true;
   672→   672→      }
   673→   673→      fn = Math.max(fn, ev.number + 1);
   674→   674→    }
   675→   675→
   676→   676→    // REBUILD THE LEDGER FROM EVENT RESULTS — the stored ledger array is
   677→   677→    // deliberately IGNORED. Older builds could append the same bout several
   678→   678→    // times (the 4x duplication bug: every fight showed as four identical
   679→   679→    // history entries and records counted 4x). Events are the source of
   680→   680→    // truth: every fought bout carries exactly ONE result object, so
   681→   681→    // rebuilding here heals any duplicated ledger on load — records,
   682→   682→    // histories, rankings and belt lineage all come out exact.
   683→   683→    for (const ev of ordered) {
   684→   684→      for (const bout of [...ev.bouts].sort((x, y) => x.order - y.order)) {
   685→   685→        if (!bout.result) continue;
   686→   686→        const a = getFighter(bout.aId);
   687→   687→        const b = getFighter(bout.bId);
   688→   688→        if (!a || !b || a.id === b.id) continue;
   689→   689→        const rec: UniverseBout = {
   690→   690→          boutId: bout.id,
   691→   691→          aId: a.id,
   692→   692→          bId: b.id,
   693→   693→          winner: bout.result.winner,
   694→   694→          method: bout.result.method,
   695→   695→          detail: bout.result.detail,
   696→   696→          round: bout.result.round,
   697→   697→          time: bout.result.time,
   698→   698→          evName: eventDisplayName(ev),
   699→   699→          section: bout.section,
   700→   700→          date: prettyDate(ev.dateISO),
   701→   701→          wc: boutWeightClass(a, b),
   702→   702→          title: !!bout.title,
   703→   703→        };
   704→   704→        applyBout(a, b, rec);
   705→   705→        transferBelts(a, b, rec.winner, rec.wc);
   706→   706→        state.ledger.push(rec);
   707→   707→      }
   708→   708→      if (!ev.completed && ev.bouts.length > 0 && ev.bouts.every((x) => x.result)) {
   709→   709→        ev.completed = true;
   710→   710→        changed = true;
   711→   711→      }
   712→   712→    }
   713→   713→    // persist the healed universe (deduped ledger + boutId keys)
   714→   714→    save();
   715→   715→    if (changed || state.ledger.length > 0) bump();
   716→   716→  } catch {
   717→   717→    state = { events: [], ledger: [] };
   718→   718→  }
   719→   719→}
