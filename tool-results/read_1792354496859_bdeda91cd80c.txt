    46→// ---------- helpers ----------
    47→function fmtSimRecord(f: Fighter): string {
    48→  const s = simRecord(f);
    49→  return `${s.w}-${s.l}-${s.d}`;

    52→function methodText(res: NonNullable<SimEventBout["result"]>): string {
    53→  const base =
    54→    res.method === "KO" ? `KO/TKO${res.detail ? ` (${res.detail})` : ""}` :
    55→    res.method === "SUB" ? `SUB${res.detail ? ` (${res.detail})` : ""}` :
    56→    res.method === "UDEC" ? "Unanimous Dec" :
    57→    res.method === "SDEC" ? "Split Dec" :
    58→    res.method === "MDEC" ? "Majority Dec" : "DQ";
    59→  return res.method === "KO" || res.method === "SUB" ? `${base} · R${res.round} ${res.time}` : base;
    60→}

    69→// settle any open wallet bets that target this pair under this event
    70→function settleBetsForBout(aId: string, bId: string, labels: string[], sim: SimResult) {
    71→  const a = getFighter(aId);
    72→  const b = getFighter(bId);
    73→  if (!a || !b) return { won: 0, lost: 0, staked: 0 };
    74→  const note = resultSummary(sim, a, b);
    75→  const winnerId = sim.winner === "A" ? aId : sim.winner === "B" ? bId : "";
    76→  let won = 0;
    77→  let lost = 0;
    78→  let staked = 0;
    79→  for (const bet of useBetStore.getState().bets) {
    80→    if (bet.status !== "open" || !labels.includes(bet.eventLabel)) continue;
    81→    if (
    82→      !((bet.bettorFighterId === aId && bet.opponentId === bId) ||
    83→        (bet.bettorFighterId === bId && bet.opponentId === aId))
    84→    ) continue;
    85→    const isWin = bet.bettorFighterId === winnerId;
    86→    const payout = bet.stake * (bet.decOdds - 1);
    87→    useBetStore.getState().settleBet(bet.id, isWin, note, payout);
    88→    if (isWin) {
    89→      won++;
    90→    } else {
    91→      lost++;
    92→      staked += bet.stake;
    93→    }
    94→  }
    95→  return { won, lost, staked };
    96→}

   434→function EventCard({ ev }: { ev: SimEvent }) {
   435→  const { setSimSpec } = useApp();
   436→  const { bets } = useBetStore();
   437→  const name = eventDisplayName(ev);
   438→  const tagline = eventTagline(ev);
   439→  const pending = ev.bouts.filter((b) => !b.result);
   440→  const openBetsCount = bets.filter(
   441→    (b) => b.status === "open" && b.eventLabel === name
   442→  ).length;
   443→
   444→  const runFullCard = () => {
   445→    if (pending.length === 0) return;
   446→    // broadcast order: early prelims → prelims → main card
   447→    const order: EventSection[] = ["early", "prelim", "main"];
   448→    let won = 0;
   449→    let lost = 0;
   450→    let staked = 0;
   451→    let belts = 0;
   452→    for (const section of order) {
   453→      for (const bout of eventBouts(ev, section)) {
   454→        if (bout.result) continue;
   455→        const a = getFighter(bout.aId)!;
   456→        const b = getFighter(bout.bId)!;
   457→        if (!a || !b) continue;
   458→        const sim = simulateFight(a, b, bout.rounds);
   459→        const outcome = recordEventBout(ev.id, bout.id, sim);
   460→        if (outcome) {
   461→          if (outcome.belt) {
   462→            belts++;
   463→            toast.success(
   464→              `NEW CHAMPION! ${outcome.belt.newChamp.name} dethrones ${outcome.belt.oldChamp.name} — the ${outcome.belt.divName} belt changes hands!`,
   465→              { duration: 7000 }
   466→            );
   467→          }
   468→          const s = settleBetsForBout(a.id, b.id, eventSettleLabels(ev), sim);
   469→          won += s.won;
   470→          lost += s.lost;
   471→          staked += s.staked;
   472→        }
   473→      }
   474→    }
   475→    toast.success(
   476→      ev.completed
   477→        ? `${name} is in the books — records, rankings & titles updated. Next fight week ${prettyDate(isoOf(currentSaturday()))} unlocked.`

   603→// ---------- past event row ----------
   604→function PastEventRow({ ev }: { ev: SimEvent }) {
   605→  const [open, setOpen] = useState(false);
   606→  const name = eventDisplayName(ev);
   607→  const mainBout = eventBouts(ev, "main")[0];
   608→  const mainA = mainBout ? getFighter(mainBout.aId) : null;
   609→  const mainB = mainBout ? getFighter(mainBout.bId) : null;
   610→  const mainWin = mainBout?.result
   611→    ? mainBout.result.winner === "A" ? mainA : mainBout.result.winner === "B" ? mainB : null
   612→    : null;
   613→
   614→  return (
   615→    <div className="overflow-hidden rounded-xl border border-white/10 bg-white/[0.015]">
   616→      <button onClick={() => setOpen(!open)} className="flex w-full items-center gap-3 px-3.5 py-3 text-left transition-colors hover:bg-white/[0.03]">
   617→        <div className="min-w-0 flex-1">
   618→          <div className="flex items-center gap-2">
   619→            <span className={`text-[9px] font-display font-bold tracking-widest ${ev.type === "numbered" ? "text-[#e0b64c]" : "text-sky-400"}`}>
   620→              {ev.type === "numbered" ? `UFC ${ev.number}` : `FIGHT NIGHT${ev.number ? ` ${ev.number}` : ""}`}
   621→            </span>
   622→            <span className="num text-[10px] text-zinc-600">{prettyDate(ev.dateISO)}</span>
   623→          </div>
   624→          <div className="truncate font-display text-sm font-bold text-white">{name}</div>
   625→          {mainWin && mainA && mainB && (
   626→            <div className="truncate text-[10px] text-zinc-500">
   627→              {mainWin.name} def. {mainWin === mainA ? mainB.name : mainA.name}
   628→              {mainBout?.result && ` · ${methodText(mainBout.result)}`}
   629→            </div>
   630→          )}
   631→        </div>
   632→        <Trophy className="h-4 w-4 shrink-0 text-zinc-600" />
   633→        <ChevronDown className={`h-4 w-4 shrink-0 text-zinc-600 transition-transform ${open ? "" : "-rotate-90"}`} />
   634→      </button>
   635→      {open && (
   636→        <div className="space-y-1.5 border-t border-white/5 px-3.5 py-3">
   637→          {(["early", "prelim", "main"] as EventSection[]).flatMap((section) =>
   638→            eventBouts(ev, section).map((bout) => {
   639→              const a = getFighter(bout.aId);
   640→              const b = getFighter(bout.bId);
   641→              if (!a || !b || !bout.result) return null;
   642→              const w = bout.result.winner === "A" ? a : bout.result.winner === "B" ? b : null;
   643→              return (
   644→                <div key={bout.id} className="flex items-center justify-between gap-2 rounded-lg bg-white/[0.02] px-2.5 py-1.5 text-[11px]">
   645→                  <span className="shrink-0 text-[8px] font-display tracking-widest text-zinc-600">
   646→                    {boutSectionLabel(ev, bout)}
   647→                  </span>
   648→                  <span className="min-w-0 flex-1 truncate text-zinc-400">
   649→                    <span className="font-semibold text-zinc-200">{w ? w.name : "Draw"}</span>
   650→                    {w ? ` def. ${w === a ? b.name : a.name}` : ` — ${a.name} vs ${b.name}`}
   651→                  </span>
   652→                  <span className="shrink-0 text-[10px] text-zinc-500">{methodText(bout.result)}</span>
   653→                </div>
   654→              );
   655→            })
   656→          )}
   657→        </div>
   658→      )}
   659→    </div>
   660→  );
   661→}
